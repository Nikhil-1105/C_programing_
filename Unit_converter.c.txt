c#include <stdio.h>

int main() {
    // Setting the Fahrenheit value to exactly 98.6
    float fahrenheit = 98.6;
    float celsius;

    // Conversion formula
    celsius = (fahrenheit - 32.0) * 5.0 / 9.0;

    // Displaying the results rounded to 2 decimal places
    printf("Enter temperature in Fahrenheit: 98.6\n");
    printf("\n--- Conversion Result ---\n");
    printf("Fahrenheit: %.2f F\n", fahrenheit);
    printf("Celsius:    %.2f C\n", celsius);

    return 0;
}
